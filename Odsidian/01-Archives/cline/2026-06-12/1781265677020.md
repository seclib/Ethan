 string    // \"low\" | \"medium\" | \"high\" | \"critical\"\n    Signals    []Signal  // Each contributing factor\n}\n\ntype Signal struct {\n    Source   string\n    Weight   float64\n    Score    float64\n    Evidence string\n    Timestamp time.Time\n}\n\nfunc (e *ThreatEngine) CalculateScore(entity *Entity) ThreatScore {\n    signals := e.gatherSignals(entity)\n    \n    // Multi-source weighted average\n    var totalWeight, totalScore float64\n    for _, s := range signals {\n        totalWeight += s.Weight\n        totalScore += s.Score * s.Weight\n    }\n    \n    confidence := totalScore / totalWeight\n    \n    // Temporal decay : older signals weigh less\n    for i := range signals {\n        age := time.Since(signals[i].Timestamp)\n        decay := math.Exp(-age.Hours() / 720.0) // 30-day half-life\n        signals[i].Weight *= decay\n    }\n    \n    // Context boost : co-occurrence with known bad entities\n    contextBoost := e.checkContextCorrelation(entity)\n    confidence = math.Min(1.0, confidence * (1.0 + contextBoost))\n    \n    return ThreatScore{\n        Confidence: confidence,\n        Severity:   e.classifySeverity(confidence),\n        Priority:   e.determinePriority(confidence),\n        Signals:    signals,\n    }\n}\n```\n\n---\n\n## 12. string\n    WorkspaceID string\n    UserID      string\n    Timestamp   time.Time\n    Payload     any\n    Metadata    map[string]string\n}\n\ntype Handler func(context.Context, Event) error\n\ntype InProcessBus struct {\n    handlers map[string][]Handler\n    mu       sync.RWMutex\n}\n\nfunc New() *InProcessBus {\n    return &InProcessBus{\n        handlers: make(map[string][]Handler),\n    }\n}\n\nfunc (b *InProcessBus) Subscribe(topic string, handler Handler) func() {\n    b.mu.Lock()\n    b.handlers[topic] = append(b.handlers[topic], handler)\n    b.mu.Unlock()\n    \n    // Return unsubscribe function\n    return func() {\n        b.mu.Lock()\n        defer b.mu.Unlock()\n        handlers := b.handlers[topic]\n        for i, h := range handlers {\n            if &h == &handler { // pointer comparison\n                b.handlers[topic] = append(handlers[:i], handlers[i+1:]...)\n                break\n            }\n        }\n    }\n}\n\nfunc (b *InProcessBus) Publish(ctx context.Context, event Event) error {\n    b.mu.RLock()\n    handlers := b.handlers[event.Type]\n    b.mu.RUnlock()\n    \n    var g errgroup.Group\n    for _, h := range handlers {\n        h := h\n        g.Go(func() error {\n            return h(ctx, event)\n        })\n    }\n    return g.Wait()\n}\n```\n\n### 4.4 🔌 Plugin Engine (WASM Sandbox)\n\n#### 4.4.1 State
